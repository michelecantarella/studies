/**
 * Google Apps Script backend — WT Study, build WT_S (keys and cards)
 *
 * Deploy this as its OWN Apps Script project, bound to its own spreadsheet —
 * never on top of another study's sheet (ensureSheet_ rewrites a sheet's
 * header row when it doesn't match).
 *
 * Sheets:
 *   Meta      — one row per respondent (study-level + resume snapshot)
 *   Responses — one row per respondent × sequence × attempt. Everything
 *               about a sequence — its draw, the offer, the switch, the
 *               outcome — lives on this one row.
 *   Seeds     — the pool of quartets (full / mixed / trunc / trunc_gift).
 *
 * Deploy: Extensions → Apps Script → Deploy → New deployment
 *   Execute as: Me  |  Who has access: Anyone
 *
 * GET  ?action=lookup&pid=XXX
 *   → { found, study_complete, resume_snapshot }
 *
 * POST { action:'progress', pid, study_id, session_id, study_level, sections, snapshot }
 *   → { ok: true }   (non-blocking: tryLock 1 s, skipped if busy)
 *
 * POST { action:'complete', pid, study_id, session_id, study_level, sections }
 *   → { ok: true }   (blocking: waitLock 10 s, idempotent)
 */

// ── Sheet helpers ────────────────────────────────────────────────

function ensureSheet_(name, header) {
  var ss    = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName(name);
  if (!sheet) sheet = ss.insertSheet(name);
  var current = sheet.getLastColumn() > 0
    ? sheet.getRange(1, 1, 1, Math.max(sheet.getLastColumn(), header.length)).getValues()[0]
    : [];
  var matches = current.length >= header.length &&
    header.every(function(h, i) { return current[i] === h; });
  if (!matches) {
    sheet.getRange(1, 1, 1, header.length).setValues([header]);
    sheet.setFrozenRows(1);
  }
  return sheet;
}

function getMetaSheet_()  { return ensureSheet_(META_SHEET, META_HEADER); }
function getRespSheet_()  { return ensureSheet_(RESP_SHEET, RESP_HEADER); }
function getSeedSheet_()  { return ensureSheet_(SEED_SHEET, SEED_HEADER); }

// WT_S (2026-11-05): a sheet cell holds at most 50,000 characters, and a longer one makes the whole row's write fail.
// Long texts are split over several columns (CELL_MAX characters each); anything still longer is cut (cellValue_).
var CELL_MAX = 45000;
var SNAP_COLS_ = ['resume_snapshot_json', 'resume_snapshot_json_2', 'resume_snapshot_json_3', 'resume_snapshot_json_4', 'resume_snapshot_json_5', 'resume_snapshot_json_6', 'resume_snapshot_json_7', 'resume_snapshot_json_8'];
var TASKS_COLS_ = ['tasks_json', 'tasks_json_2', 'tasks_json_3', 'tasks_json_4'];
function splitCell_(s, n) {
  var out = []; for (var k = 0; k < n; k++) out.push(s.slice(k * CELL_MAX, (k + 1) * CELL_MAX)); return out;
}
function cellValue_(v) {
  var c = cellValue0_(v);
  return typeof c === 'string' && c.length > 49990 ? c.slice(0, 49990) : c;
}
function cellValue0_(v) {
  if (v === null || v === undefined) return '';
  if (typeof v === 'object') return JSON.stringify(v);
  // Strings made only of numbers must not be reinterpreted by the sheet's locale
  // (with an Italian locale "2,5" becomes 2.5, "4,24,30" a date, and "0.638" a
  // time, 638 minutes): a single number is written as a number, a list of numbers
  // (joined by "," or "|", e.g. winning_cards "4,24,30", deck_d1_options
  // "0.638|0.41") as text. The leading apostrophe is not stored in the value.
  if (typeof v === 'string') {
    if (/^-?\d+(\.\d+)?$/.test(v)) return Number(v);
    if (/^[0-9.,|\s\-]+$/.test(v) && /[.,|]/.test(v)) return "'" + v;
  }
  return v;
}

// Write (or update) a single row using a header array to map field names → column indices.
function writeRowBatched_(sheet, rowIndex, isNew, updates, header) {
  var row = isNew
    ? new Array(header.length).fill('')
    : (function() {
        var r = sheet.getRange(rowIndex, 1, 1, header.length).getValues()[0];
        while (r.length < header.length) r.push('');
        return r;
      })();
  updates.forEach(function(pair) {
    var col = header.indexOf(pair[0]);
    if (col !== -1) row[col] = cellValue_(pair[1]);
  });
  sheet.getRange(rowIndex, 1, 1, header.length).setValues([row]);
}

// ── Meta row lookup ──────────────────────────────────────────────

function findMetaRow_(sheet, pid) {
  var lastRow = sheet.getLastRow();
  if (lastRow < 2) return null;
  var pidCol = META_HEADER.indexOf('prolific_pid');
  var data   = sheet.getRange(2, 1, lastRow - 1, pidCol + 1).getValues();
  for (var i = 0; i < data.length; i++) {
    if (String(data[i][pidCol]) === String(pid)) return i + 2;
  }
  return null;
}

function readMetaRow_(sheet, rowIndex) {
  return sheet.getRange(rowIndex, 1, 1, META_HEADER.length).getValues()[0];
}

// ── Response row lookup ──────────────────────────────────────────
// Returns a map of seq_n (as string) → { attemptNumber → rowIndex } for the
// given pid. Rows written before seq_attempt existed have no value there —
// treated as attempt 1, same as a fresh single-attempt section.

function findResponseRows_(sheet, pid) {
  var lastRow = sheet.getLastRow();
  if (lastRow < 2) return {};
  var pidCol = RESP_HEADER.indexOf('prolific_pid');
  var seqCol = RESP_HEADER.indexOf('seq_n');
  var attCol = RESP_HEADER.indexOf('seq_attempt');
  var nCols  = Math.max(pidCol, seqCol, attCol) + 1;
  var data   = sheet.getRange(2, 1, lastRow - 1, nCols).getValues();
  var map    = {};
  for (var i = 0; i < data.length; i++) {
    if (String(data[i][pidCol]) !== String(pid)) continue;
    var seqKey = String(data[i][seqCol]);
    var att    = Number(data[i][attCol]) || 1;
    if (!map[seqKey]) map[seqKey] = {};
    map[seqKey][att] = i + 2;
  }
  return map;
}

// WT_N (2026-10-29): a progress write is ignored when the study is already complete,
// or when it comes from a stale page (fewer sequences done than already stored).
function staleProgress_(sheet, pid, sl) {
  var rowIndex = findMetaRow_(sheet, pid);
  if (!rowIndex) return false;
  var row = readMetaRow_(sheet, rowIndex);
  if (row[META_HEADER.indexOf('study_complete_at')]) return true;
  var stored = Number(row[META_HEADER.indexOf('sections_done')] || 0);
  return sl.sections_done != null && Number(sl.sections_done) < stored;
}

// ── Upsert Responses rows ─────────────────────────────────────────
// Writes one row per section in the sections array. Each (seq_n, seq_attempt)
// pair gets its own row — a new-session reload that discarded an in-progress
// attempt (see index.html's applyResume_/buildAbandonedRecord_) no longer
// overwrites it, it just adds attempt 2 (3, ...) alongside it. After writing,
// is_latest_attempt is recomputed for every seq_n touched so it always
// reflects the true max attempt on the sheet, not just what's in this batch.

function upsertRespRows_(sheet, pid, study_id, session_id, demo_mode, sections) {
  if (!sections || !sections.length) return;
  var rowMap     = findResponseRows_(sheet, pid); // seq_n -> {attempt: rowIndex}
  var touchedSeq = {};
  // Filled here from the request itself, never from a section's own fields;
  // is_latest_attempt is recomputed below.
  var SERVER_COLS = { prolific_pid:1, study_id:1, session_id:1, demo_mode:1, seq_attempt:1, is_latest_attempt:1 };

  for (var i = 0; i < sections.length; i++) {
    var s        = sections[i];
    var seqKey   = String(s.seq_n);
    var attempt  = s.seq_attempt != null ? Number(s.seq_attempt) : 1;
    if (!rowMap[seqKey]) rowMap[seqKey] = {};
    var rowIndex = rowMap[seqKey][attempt];
    var isNew    = !rowIndex;
    if (isNew) rowIndex = sheet.getLastRow() + 1;

    var updates = [
      ['prolific_pid', pid],
      ['study_id',     study_id   || ''],
      ['session_id',   session_id || ''],
      ['demo_mode',    demo_mode  != null ? demo_mode : ''],
      ['seq_attempt',  attempt],
    ];
    // Every other column is copied by name from the section object the
    // client sent (null/undefined → blank, objects/arrays → JSON string —
    // see cellValue_). A column the client didn't send is left untouched.
    RESP_HEADER.forEach(function(col) {
      if (SERVER_COLS[col] || TASKS_COLS_.indexOf(col) >= 0) return;
      if (Object.prototype.hasOwnProperty.call(s, col)) updates.push([col, s[col]]);
    });
    // tasks_json, split over TASKS_COLS_ (read back joined)
    if (Object.prototype.hasOwnProperty.call(s, 'tasks_json')) {
      var tj = typeof s.tasks_json === 'string' ? s.tasks_json : JSON.stringify(s.tasks_json || []);
      var tParts = splitCell_(tj, TASKS_COLS_.length);
      TASKS_COLS_.forEach(function(c, k) { updates.push([c, tParts[k]]); });
    }

    writeRowBatched_(sheet, rowIndex, isNew, updates, RESP_HEADER);
    rowMap[seqKey][attempt] = rowIndex;
    touchedSeq[seqKey] = true;
  }

  // is_latest_attempt must reflect the true max attempt on the sheet for
  // each seq_n touched — not just what arrived in this call, since an
  // abandoned attempt and its successor can be sent in separate
  // progress/complete calls.
  var isLatestCol = RESP_HEADER.indexOf('is_latest_attempt') + 1;
  Object.keys(touchedSeq).forEach(function(seqKey) {
    var attempts = rowMap[seqKey]; // {attemptNumber: rowIndex}
    var attemptNums = Object.keys(attempts).map(Number);
    var maxAttempt  = Math.max.apply(null, attemptNums);
    attemptNums.forEach(function(att) {
      var cell = sheet.getRange(attempts[att], isLatestCol);
      var want = (att === maxAttempt) ? 1 : 0;
      if (cell.getValue() !== want) cell.setValue(want);
    });
  });
}

// ── WT_M: pairing pool ────────────────────────────────────────────
function seedRows_(sheet) {
  var lastRow = sheet.getLastRow();
  if (lastRow < 2) return [];
  var vals = sheet.getRange(2, 1, lastRow - 1, SEED_HEADER.length).getValues();
  return vals.map(function(r, i) { var o = { _row: i + 2 }; SEED_HEADER.forEach(function(h, j) { o[h] = r[j]; }); return o; });
}
function seedSet_(sheet, row, fields) {
  Object.keys(fields).forEach(function(k) {
    var c = SEED_HEADER.indexOf(k); if (c >= 0) sheet.getRange(row, c + 1).setValue(fields[k]);
  });
}
function seedOpen_(r, v, nowMs) {
  var st = r[v + '_status'];
  if (st === 'open') return true;
  if (st === 'claimed') { var t = Date.parse(r[v + '_at']); return !(t > 0) || nowMs - t > SEED_CLAIM_HOURS * 3600 * 1000; }
  return false;
}
// Sequences from the unpaid test modes (STUDY_ID pilot / studentpilot) are never pooled.
function isPilotStudy_(study_id) { return /^(pilot|studentpilot)$/i.test(String(study_id || '')); }
// One claim per requested slot {version}: an open version (the one asked
// for, or any with 'any') of a design made by someone else and not already
// played by this pid, from any study except the test modes; preferring designs
// with more of their other versions already done, then the oldest. Partners are
// unique: the claimant is never in two quartets with the same person (in this
// claim, or in quartets already formed in the sheet). Returns one entry per slot
// (null = none open).
function seedClaim_(pid, slots, now) {
  var sheet = getSeedSheet_(), rows = seedRows_(sheet), nowMs = Date.parse(now), out = [], used = {};
  // the other participants of a design (its creator and whoever holds its other versions)
  var partnersOf = function(r, v) { var p = []; if (r.created_by) p.push(String(r.created_by));
    SEED_VERSIONS.forEach(function(o) { if (o !== v && r[o + '_pid']) p.push(String(r[o + '_pid'])); }); return p; };
  var partners = {};
  rows.forEach(function(r) {
    SEED_VERSIONS.forEach(function(v) { if (String(r[v + '_pid']) === String(pid)) partnersOf(r, v).forEach(function(q) { partners[q] = true; }); });
  });
  (slots || []).forEach(function(slot) {
    var best = null, bestScore = -1, want = slot && slot.version;
    rows.forEach(function(r) {
      if (used[r._row] || isPilotStudy_(r.study_id)) return;
      if (String(r.created_by) === String(pid)) return;
      if (SEED_VERSIONS.some(function(o) { return String(r[o + '_pid']) === String(pid); })) return; // one version per design per person
      SEED_VERSIONS.forEach(function(v) {
        if (want && want !== 'any' && v !== want) return;
        if (!seedOpen_(r, v, nowMs)) return;
        if (partnersOf(r, v).some(function(q) { return q !== String(pid) && partners[q]; })) return;
        var done = SEED_VERSIONS.filter(function(o) { return o !== v && r[o + '_status'] === 'done'; }).length;
        var score = (done + 1) * 1e13 - (Date.parse(r.created_at) || 0);
        if (score > bestScore) { bestScore = score; best = { r: r, v: v }; }
      });
    });
    if (!best) { out.push(null); return; }
    used[best.r._row] = true;
    partnersOf(best.r, best.v).forEach(function(q) { partners[q] = true; });
    var f = {}; f[best.v + '_status'] = 'claimed'; f[best.v + '_pid'] = pid; f[best.v + '_at'] = now;
    seedSet_(sheet, best.r._row, f);
    var design = null; try { design = JSON.parse(best.r.design_json); } catch (e) {}
    out.push({ seed_id: best.r.seed_id, version: best.v, design: design });
  });
  return out;
}
// Open versions still waiting for a partner, by version (for planning a clearing batch).
function seedPoolStatus_(now) {
  var rows = seedRows_(getSeedSheet_()), nowMs = Date.parse(now), out = { open: 0, by_version: { full: 0, mixed: 0, trunc: 0, trunc_gift: 0 } };
  rows.forEach(function(r) {
    if (isPilotStudy_(r.study_id)) return;
    SEED_VERSIONS.forEach(function(v) {
      if (!seedOpen_(r, v, nowMs)) return;
      out.open++; out.by_version[v]++;
    });
  });
  return out;
}
function seedAdd_(pid, study_id, seeds, now) {
  if (isPilotStudy_(study_id)) return; // test modes never pooled
  var sheet = getSeedSheet_();
  (seeds || []).forEach(function(s) {
    var row = new Array(SEED_HEADER.length).fill('');
    var put = function(k, v) { row[SEED_HEADER.indexOf(k)] = v; };
    put('seed_id', s.seed_id); put('study_id', study_id || ''); put('created_at', now); put('created_by', pid);
    put('design_json', JSON.stringify(s.design));
    SEED_VERSIONS.forEach(function(v) {
      if (v === s.version) { put(v + '_status', 'claimed'); put(v + '_pid', pid); put(v + '_at', now); }
      else put(v + '_status', 'open');
    });
    sheet.appendRow(row);
  });
}
function seedMark_(pid, seed_id, version, status, now) {
  if (SEED_VERSIONS.indexOf(version) < 0) return;
  var sheet = getSeedSheet_(), rows = seedRows_(sheet);
  for (var i = 0; i < rows.length; i++) {
    var r = rows[i]; if (String(r.seed_id) !== String(seed_id)) continue;
    var f = {};
    if (status === 'done') { f[version + '_status'] = 'done'; f[version + '_pid'] = pid; f[version + '_at'] = now; }
    else if (String(r[version + '_pid']) === String(pid) && r[version + '_status'] === 'claimed') { f[version + '_status'] = 'open'; f[version + '_pid'] = ''; f[version + '_at'] = ''; }
    seedSet_(sheet, r._row, f);
    return;
  }
}

// ── HTTP handlers ────────────────────────────────────────────────

function doGet(e) {
  var lock = LockService.getScriptLock();
  lock.waitLock(10000);
  try {
    var action = e.parameter.action;
    if (action === 'pool') return jsonOut_(seedPoolStatus_(new Date().toISOString())); // ?action=pool: open versions waiting for a partner
    if (action !== 'lookup') return jsonOut_({ error: 'unknown action' });

    var pid = e.parameter.pid;
    if (!pid) return jsonOut_({ error: 'missing pid' });

    var sheet    = getMetaSheet_();
    var rowIndex = findMetaRow_(sheet, pid);

    if (!rowIndex) {
      return jsonOut_({ found: false, study_complete: false, resume_snapshot: null });
    }

    var row           = readMetaRow_(sheet, rowIndex);
    var get           = function(f) { return row[META_HEADER.indexOf(f)]; };
    var studyComplete = !!get('study_complete_at');
    var resumeSnap    = null;

    if (!studyComplete) {
      try {
        var raw = SNAP_COLS_.map(function(c) { return get(c) || ''; }).join('');
        if (raw) resumeSnap = JSON.parse(raw);
      } catch (err) {}
    }

    return jsonOut_({
      found:           true,
      study_complete:  studyComplete,
      resume_snapshot: resumeSnap
    });
  } finally {
    lock.releaseLock();
  }
}

function doPost(e) {
  var body       = JSON.parse(e.postData.contents);
  var pid        = body.pid;
  if (!pid) return jsonOut_({ ok: false, error: 'missing pid' });

  var now        = new Date().toISOString();
  var sl         = body.study_level || {};
  var sections   = body.sections    || [];
  var study_id   = body.study_id    || '';
  var session_id = body.session_id  || '';

  // ── WT_M: pairing pool (blocking, short) ─────────────────────
  if (body.action === 'seedClaim' || body.action === 'seedAdd' || body.action === 'seedRelease' || body.action === 'seedDone') {
    var lockS = LockService.getScriptLock();
    lockS.waitLock(10000);
    try {
      if (body.action === 'seedClaim') {
        if (isPilotStudy_(study_id)) return jsonOut_({ ok: true, seeds: [] });
        var slots = body.slots || [];
        return jsonOut_({ ok: true, seeds: seedClaim_(pid, slots, now) });
      }
      if (body.action === 'seedAdd') { seedAdd_(pid, study_id, body.seeds, now); return jsonOut_({ ok: true }); }
      seedMark_(pid, body.seed_id, body.version, body.action === 'seedDone' ? 'done' : 'release', now);
      return jsonOut_({ ok: true });
    } catch (err) {
      return jsonOut_({ ok: false, error: String(err) });
    } finally {
      lockS.releaseLock();
    }
  }

  // ── progress (non-blocking) ──────────────────────────────────
  if (body.action === 'progress') {
    var lock    = LockService.getScriptLock();
    var gotLock = lock.tryLock(1000);
    if (!gotLock) return jsonOut_({ ok: true, skipped: true });
    try {
      var metaSheet  = getMetaSheet_();
      var respSheet  = getRespSheet_();
      if (staleProgress_(metaSheet, pid, sl)) return jsonOut_({ ok: true, ignored: true });
      // WT_S (2026-11-05): the sequences first, so that a failure writing Meta (e.g. the snapshot) cannot lose them
      upsertRespRows_(respSheet, pid, study_id, session_id, sl.demo_mode, sections);
      upsertMeta_(metaSheet, pid, study_id, session_id, sl, body.snapshot, now, false);
      return jsonOut_({ ok: true });
    } catch (err) {
      return jsonOut_({ ok: false, error: String(err) });
    } finally {
      lock.releaseLock();
    }
  }

  // ── complete (blocking, idempotent) ─────────────────────────
  if (body.action === 'complete') {
    var lock2 = LockService.getScriptLock();
    lock2.waitLock(10000);
    try {
      var metaSheet2  = getMetaSheet_();
      var respSheet2  = getRespSheet_();

      // Idempotent: already complete → return ok without re-writing
      var existingRow = findMetaRow_(metaSheet2, pid);
      if (existingRow) {
        var existingMeta = readMetaRow_(metaSheet2, existingRow);
        if (existingMeta[META_HEADER.indexOf('study_complete_at')]) {
          return jsonOut_({ ok: true, already_complete: true });
        }
      }

      upsertRespRows_(respSheet2, pid, study_id, session_id, sl.demo_mode, sections);
      upsertMeta_(metaSheet2, pid, study_id, session_id, sl, null, now, true);
      return jsonOut_({ ok: true });
    } catch (err) {
      return jsonOut_({ ok: false, error: String(err) });
    } finally {
      lock2.releaseLock();
    }
  }

  return jsonOut_({ ok: false, error: 'unknown action' });
}

function jsonOut_(obj) {
  return ContentService.createTextOutput(JSON.stringify(obj))
    .setMimeType(ContentService.MimeType.JSON);
}
