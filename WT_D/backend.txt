/**
 * Google Apps Script backend — WT Study v2
 *
 * Two sheets:
 *   Meta      — one row per respondent (study-level + resume snapshot)
 *   Responses — one row per respondent × sequence (long-form panel data)
 *
 * Deploy: Extensions → Apps Script → Deploy → New deployment
 *   Execute as: Me  |  Who has access: Anyone
 *
 * GET  ?action=lookup&pid=XXX
 *   → { found, study_complete, resume_snapshot }
 *
 * POST { action:'progress', pid, study_id, session_id, study_level, sections, snapshot }
 *   → { ok: true }   (non-blocking: tryLock 1 s, skipped if busy)
 *
 * POST { action:'complete', pid, study_id, session_id, study_level, sections }
 *   → { ok: true }   (blocking: waitLock 10 s, idempotent)
 */

var RESP_HEADER = [
  'prolific_pid', 'study_id', 'session_id', 'demo_mode',
  'seq_n',
  'pay', 'pay_tasks', 'regime',
  // FIRST hand's disclosed structure: hand_len (total tasks), live_pattern
  // (one char per task, in order — L=live/can-end, S=safe/cannot; e.g.
  // "SSLSLS" — MIXED throughout, not a block of S then a block of L),
  // p_inside_first (its end-probability). card_order is the FULL sequence's
  // card-by-card order across every hand actually played (see below).
  // hands_json is the full realised chain (every hand's own pattern + which
  // sub-regime placed its offer); n_hands is how many hands the section
  // actually ran; n_req is the total main-sequence task count.
  'hand_len', 'live_pattern', 'card_order', 'p_inside_first', 'n_hands', 'n_req',
  'pause', 'pause_max', 'alt_duration', 'alt_end', 'pause_secs',
  'outcome', 'switch_offered', 'switch_at',
  'n_tasks_done', 'earnings', 'grid_pay',
  'total_targets', 'total_found', 'total_false_pos', 'total_missed',
  'seq_reload_count', 'start_at', 'end_at',
  'tasks_json',
  'seq_frame_treatment', 'seq_is_bonus',
  'switch_offered_at_task', 'switch_offered_at_time',
  'switch_taken', 'switch_taken_at_task', 'switch_taken_at_time',
  'forfeit_taken', 'forfeit_taken_at_task', 'forfeit_taken_at_time',
  'seq_start_at_time', // explicit alias of start_at
  // ── Looping-"hand" design fields (MIXED) ──────────────────────────
  //
  // The main sequence is a chain of hands. UNLIKE the other builds, a hand
  // is NOT a block of safe cards followed by a block of live cards — every
  // task in it is independently, randomly marked live (can end the hand) or
  // safe (cannot), so the two are interspersed throughout (see live_pattern
  // / hands_json[].live_pattern). Each hand has its own disclosed `p_inside`
  // (chance the sequence ends inside it); if it ends, the ending card is
  // uniform over whichever tasks were marked live; if it survives, a fresh
  // hand loops in (new length, new pattern, new p_inside). hands_json is the
  // full realised chain (one object per hand: hand_len, live_pattern,
  // p_inside, sub_regime, wasted, ends, end_pos, played_len, pause, plus
  // pause-context bookkeeping — active_passed_in_hand/_in_seq and
  // inactive_passed_in_hand/_in_seq count live/safe cards already gone by AT
  // that hand's own pause, within just that hand vs. across the whole
  // sequence so far; the _left_ variants count how many remain, within that
  // hand vs. across the rest of the realised chain). card_order
  // concatenates every hand's live_pattern, TRUNCATED to played_len (cards
  // after an early end were never shown), joined with "|" — the exact
  // card-by-card order presented across the whole main sequence. The last
  // hand in hands_json is the ending one.
  //
  // regime (above) is the META regime, fixed PER RESPONDENT, and governs how
  // each hand's offer (pause) is placed relative to ITS OWN ending:
  //   1 — every hand: ending drawn first, offer after. LEAKS: E[n_req|pause]
  //       rises with pause.
  //   2 — every hand: offer first, ending conditional on surviving it. Flat,
  //       never wastes the offer.
  //   3 — every hand: pause and ending drawn FULLY INDEPENDENTLY. Flat like
  //       2, but can WASTE the offer (hands_json[].wasted=1) if the hand
  //       already ended at or before its own pause card.
  //   0 — the "mixed" regime: EACH HAND independently re-rolls which of
  //       {1,2} governs it (never 3), rather than the whole respondent being
  //       fixed to one. hands_json[].sub_regime records which one actually
  //       applied to that specific hand (constant across all hands, equal to
  //       regime, when regime is 1, 2, or 3; varies hand-to-hand only when
  //       regime=0).
  //
  // e_end_at_start: expected TOTAL main length at the start (first-hand
  // actuals + population means for later hands). e_remaining = e_end_at_start
  // - pause; e_end_at_pause = pause + e_remaining. pay_tasks is drawn around
  // e_end_at_start; alt_duration is drawn around e_remaining (so the
  // alternative's endpoint sits at e_end_at_start). alt_base = e_end_at_start,
  // alt_width = the mean hand length (the beta-normal support width).
  // alt_minus_expected = alt_duration - e_remaining (negative = alt shorter).
  'e_remaining', 'e_end_at_start', 'e_end_at_pause',
  // Mistake penalty accrued during this section (£0.01 per missed/false-
  // positive click), deducted from earnings.
  'penalty',
  // 1 iff the UNDISCLOSED hand cap (index.html's CFG.maxHands, currently 6)
  // forced the last hand to end rather than the section ending on its own
  // draw — treat these rows as right-censored at n_req.
  'cap_bound',
  // 1 iff ANY hand in this section wasted its own offer (only reachable when
  // that hand's sub_regime was 3) — see hands_json[].wasted for exactly
  // which hand(s). No switch decision exists for a wasted hand's own pause.
  'offer_wasted',
  'alt_base', 'alt_width', 'alt_minus_expected',
  // Full realised chain of hands (JSON array — one object per hand).
  'hands_json',
  // switch_pressed_at_task/_time: when "Switch to alternative" was pressed
  // (opening the confirm dialog) — distinct from switch_taken_at_task/_time
  // (when "Confirm switch" was clicked). switch_decision_secs: elapsed
  // seconds between them. All three only populated if switch_taken=1.

  // alt_phase_started_at: when the alternative sequence actually began.

  // outcome_summary: a fixed human-readable label, derived from the columns
  // above (never a new source of truth) — see computeOutcomeSummary_() in
  // index.html.
  'outcome_summary',
  // ── Reload/attempt tracking ────────────────────────────────────────
  //
  // Historically a new-session reload mid-section (see index.html's
  // applyResume_) silently discarded that section's already-drawn hands and
  // redrew fresh ones — the abandoned hands never reached this sheet at all,
  // since in-progress sections are only ever logged once endTask() completes
  // them. Now every attempt at a given seq_n gets its OWN row instead of one
  // overwriting the next:
  //
  // seq_attempt: 1 for a respondent's first (and usually only) attempt at
  //   this seq_n; bumped to 2, 3, ... each time a new-session reload
  //   discarded an in-progress attempt with real participant progress on it
  //   (see index.html's buildAbandonedRecord_). An abandoned attempt's row
  //   has outcome='abandoned_reload' (see outcome_summary above) and
  //   whatever hands/card_order/regime data it had already drawn — its
  //   switch/forfeit timing fields are blank, since abandonment happens
  //   before any of those could resolve into a real decision.
  //
  // is_latest_attempt: 1 on whichever attempt is currently the most recent
  //   for this (prolific_pid, seq_n) — computed server-side in
  //   upsertRespRows_ (never trust a client-sent value for this), so it's
  //   correct even if attempts arrive in separate progress/complete calls.
  //   Filter to is_latest_attempt=1 to get one row per real section, the
  //   same view older analyses (single-attempt-per-seq_n) assumed.
  'seq_attempt', 'is_latest_attempt'
];

// ── Sheet helpers ────────────────────────────────────────────────

function ensureSheet_(name, header) {
  var ss    = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName(name);
  if (!sheet) sheet = ss.insertSheet(name);
  var current = sheet.getLastColumn() > 0
    ? sheet.getRange(1, 1, 1, Math.max(sheet.getLastColumn(), header.length)).getValues()[0]
    : [];
  var matches = current.length >= header.length &&
    header.every(function(h, i) { return current[i] === h; });
  if (!matches) {
    sheet.getRange(1, 1, 1, header.length).setValues([header]);
    sheet.setFrozenRows(1);
  }
  return sheet;
}

function getMetaSheet_()  { return ensureSheet_(META_SHEET, META_HEADER); }
function getRespSheet_()  { return ensureSheet_(RESP_SHEET, RESP_HEADER); }

function cellValue_(v) {
  if (v === null || v === undefined) return '';
  if (typeof v === 'object') return JSON.stringify(v);
  return v;
}

// Write (or update) a single row using a header array to map field names → column indices.
function writeRowBatched_(sheet, rowIndex, isNew, updates, header) {
  var row = isNew
    ? new Array(header.length).fill('')
    : (function() {
        var r = sheet.getRange(rowIndex, 1, 1, header.length).getValues()[0];
        while (r.length < header.length) r.push('');
        return r;
      })();
  updates.forEach(function(pair) {
    var col = header.indexOf(pair[0]);
    if (col !== -1) row[col] = cellValue_(pair[1]);
  });
  sheet.getRange(rowIndex, 1, 1, header.length).setValues([row]);
}

// ── Meta row lookup ──────────────────────────────────────────────

function findMetaRow_(sheet, pid) {
  var lastRow = sheet.getLastRow();
  if (lastRow < 2) return null;
  var pidCol = META_HEADER.indexOf('prolific_pid');
  var data   = sheet.getRange(2, 1, lastRow - 1, pidCol + 1).getValues();
  for (var i = 0; i < data.length; i++) {
    if (String(data[i][pidCol]) === String(pid)) return i + 2;
  }
  return null;
}

function readMetaRow_(sheet, rowIndex) {
  return sheet.getRange(rowIndex, 1, 1, META_HEADER.length).getValues()[0];
}

// ── Response row lookup ──────────────────────────────────────────
// Returns a map of seq_n (as string) → { attemptNumber → rowIndex } for the
// given pid. Rows written before seq_attempt existed have no value there —
// treated as attempt 1, same as a fresh single-attempt section.

function findResponseRows_(sheet, pid) {
  var lastRow = sheet.getLastRow();
  if (lastRow < 2) return {};
  var pidCol = RESP_HEADER.indexOf('prolific_pid');
  var seqCol = RESP_HEADER.indexOf('seq_n');
  var attCol = RESP_HEADER.indexOf('seq_attempt');
  var nCols  = Math.max(pidCol, seqCol, attCol) + 1;
  var data   = sheet.getRange(2, 1, lastRow - 1, nCols).getValues();
  var map    = {};
  for (var i = 0; i < data.length; i++) {
    if (String(data[i][pidCol]) !== String(pid)) continue;
    var seqKey = String(data[i][seqCol]);
    var att    = Number(data[i][attCol]) || 1;
    if (!map[seqKey]) map[seqKey] = {};
    map[seqKey][att] = i + 2;
  }
  return map;
}

// ── Upsert Meta row ───────────────────────────────────────────────

function upsertMeta_(sheet, pid, study_id, session_id, sl, snapshot, now, setComplete) {

    ['total_earnings',         sl.total_earnings         != null ? sl.total_earnings         : ''],
    ['total_grid_pay',         sl.total_grid_pay         != null ? sl.total_grid_pay         : ''],
    ['total_grids_completed',  sl.total_grids_completed  != null ? sl.total_grids_completed  : ''],
    ['total_penalty',          sl.total_penalty          != null ? sl.total_penalty          : ''],
    ['reload_count',           sl.reload_count           != null ? sl.reload_count           : ''],
    ['reload_log_json',        JSON.stringify(sl.reload_log || [])],
    ['resume_snapshot_json',   setComplete ? '' : JSON.stringify(snapshot || {})],
    ['geo_country',            sl.geo_country || ''],
    ['geo_region',             sl.geo_region  || ''],
    ['geo_city',               sl.geo_city    || ''],
    ['geo_ip',                 sl.geo_ip      || ''],
    ['geo_json',               sl.geo_json    || ''],
    ['employment_status',      sl.employment_status  || ''],
    ['labor_income',           sl.labor_income        || ''],
    ['marstat',                sl.marstat             || ''],
    // hh_others can legitimately be 0 (living alone) — must check != null,
    // not `|| ''`, or a genuine 0 would be written as blank.
    ['hh_others',              sl.hh_others           != null ? sl.hh_others           : ''],
    ['family_contributors',    sl.family_contributors || ''],
    ['education',              sl.education           || ''],
    ['ends_meet_now',          sl.ends_meet_now       || ''],
    ['ends_meet_past',         sl.ends_meet_past      || ''],
    ['ends_meet_future',       sl.ends_meet_future    || ''],
    ['risk',                   sl.risk                != null ? sl.risk                : ''],
    ['patience',               sl.patience            != null ? sl.patience            : ''],
    ['geo_provider',           sl.geo_provider        || ''],
    ['pid_missing_from_link',  sl.pid_missing_from_link != null ? sl.pid_missing_from_link : ''],
    ['training_scenario_order',   sl.training_scenario_order   || ''],
    ['training_scenarios_viewed', sl.training_scenarios_viewed || ''],
  ];

    var existingConsent = isNew ? '' : readMetaRow_(sheet, rowIndex)[META_HEADER.indexOf('consent_at')];
    if (!existingConsent) updates.push(['consent_at', sl.consent_at]);
  }

  writeRowBatched_(sheet, rowIndex, isNew, updates, META_HEADER);
}

// ── Upsert Responses rows ─────────────────────────────────────────
// Writes one row per section in the sections array. Each (seq_n, seq_attempt)
// pair gets its own row — a new-session reload that discarded an in-progress
// attempt (see index.html's applyResume_/buildAbandonedRecord_) no longer
// overwrites it, it just adds attempt 2 (3, ...) alongside it. After writing,
// is_latest_attempt is recomputed for every seq_n touched so it always
// reflects the true max attempt on the sheet, not just what's in this batch.

  if (!sections || !sections.length) return;
  var rowMap     = findResponseRows_(sheet, pid); // seq_n -> {attempt: rowIndex}
  var touchedSeq = {};

  for (var i = 0; i < sections.length; i++) {
    var s        = sections[i];

    if (!rowMap[seqKey]) rowMap[seqKey] = {};
    var rowIndex = rowMap[seqKey][attempt];
    var isNew    = !rowIndex;
    if (isNew) rowIndex = sheet.getLastRow() + 1;

    var updates = [
      ['prolific_pid',    pid],
      ['study_id',        study_id   || ''],
      ['session_id',      session_id || ''],
      ['demo_mode',       demo_mode  != null ? demo_mode  : ''],
      ['seq_n',           s.seq_n],
      ['pay',             s.pay           != null ? s.pay           : ''],
      ['pay_tasks',       s.pay_tasks     != null ? s.pay_tasks     : ''],
      ['regime',          s.regime        != null ? s.regime        : ''],
      ['hand_len',        s.hand_len      != null ? s.hand_len      : ''],
      ['live_pattern',    s.live_pattern  || ''],
      ['p_inside_first',  s.p_inside_first != null ? s.p_inside_first : ''],
      ['n_hands',         s.n_hands       != null ? s.n_hands       : ''],
      ['n_req',           s.n_req         != null ? s.n_req         : ''],
      ['pause',           s.pause         != null ? s.pause         : ''],
      ['pause_max',       s.pause_max     != null ? s.pause_max     : ''],
      ['alt_duration',    s.alt_duration  != null ? s.alt_duration  : ''],
      ['alt_end',         s.alt_end       != null ? s.alt_end       : ''],
      ['pause_secs',      s.pause_secs    != null ? s.pause_secs    : ''],
      ['outcome',         s.outcome       || ''],
      ['switch_offered',  s.switch_offered != null ? s.switch_offered : ''],
      ['switch_at',       s.switch_at     != null ? s.switch_at     : ''],
      ['n_tasks_done',    s.n_tasks_done  != null ? s.n_tasks_done  : ''],
      ['earnings',        s.earnings      != null ? s.earnings      : ''],
      ['grid_pay',        s.grid_pay      != null ? s.grid_pay      : ''],
      ['total_targets',   s.total_targets  != null ? s.total_targets  : ''],
      ['total_found',     s.total_found    != null ? s.total_found    : ''],
      ['total_false_pos', s.total_false_pos!= null ? s.total_false_pos: ''],
      ['total_missed',    s.total_missed   != null ? s.total_missed   : ''],
      ['seq_reload_count', s.seq_reload_count != null ? s.seq_reload_count : ''],
      ['start_at',        s.start_at  || ''],
      ['end_at',          s.end_at    || ''],
      ['tasks_json',      JSON.stringify(s.tasks_json || [])],
      ['seq_frame_treatment', s.seq_frame_treatment != null ? s.seq_frame_treatment : ''],
      ['seq_is_bonus',        s.seq_is_bonus         != null ? s.seq_is_bonus         : ''],
      ['switch_offered_at_task',  s.switch_offered_at_task  != null ? s.switch_offered_at_task  : ''],
      ['switch_offered_at_time',  s.switch_offered_at_time  || ''],
      ['switch_taken',             s.switch_taken            != null ? s.switch_taken            : ''],
      ['switch_taken_at_task',     s.switch_taken_at_task    != null ? s.switch_taken_at_task    : ''],

      ['forfeit_taken_at_time',    s.forfeit_taken_at_time   || ''],
      ['seq_start_at_time',        s.seq_start_at_time       || ''],
      ['e_remaining',               s.e_remaining             != null ? s.e_remaining             : ''],
      ['e_end_at_start',           s.e_end_at_start          != null ? s.e_end_at_start          : ''],
      ['e_end_at_pause',           s.e_end_at_pause          != null ? s.e_end_at_pause          : ''],

      ['cap_bound',                s.cap_bound               != null ? s.cap_bound               : ''],

  var isLatestCol = RESP_HEADER.indexOf('is_latest_attempt') + 1;
  Object.keys(touchedSeq).forEach(function(seqKey) {
    var attempts = rowMap[seqKey]; // {attemptNumber: rowIndex}
    var attemptNums = Object.keys(attempts).map(Number);
    var maxAttempt  = Math.max.apply(null, attemptNums);
    attemptNums.forEach(function(att) {
      var cell = sheet.getRange(attempts[att], isLatestCol);
      var want = (att === maxAttempt) ? 1 : 0;
      if (cell.getValue() !== want) cell.setValue(want);
    });
  });
}

// ── HTTP handlers ────────────────────────────────────────────────

function doGet(e) {
  var lock = LockService.getScriptLock();
  lock.waitLock(10000);
  try {
    var action = e.parameter.action;
    if (action !== 'lookup') return jsonOut_({ error: 'unknown action' });

    var pid = e.parameter.pid;
    if (!pid) return jsonOut_({ error: 'missing pid' });

    var sheet    = getMetaSheet_();
    var rowIndex = findMetaRow_(sheet, pid);

    if (!rowIndex) {
      return jsonOut_({ found: false, study_complete: false, resume_snapshot: null });
    }

    var row           = readMetaRow_(sheet, rowIndex);
    var get           = function(f) { return row[META_HEADER.indexOf(f)]; };
    var studyComplete = !!get('study_complete_at');
    var resumeSnap    = null;

    if (!studyComplete) {
      try {
        var raw = get('resume_snapshot_json');
        if (raw) resumeSnap = JSON.parse(raw);
      } catch (err) {}
    }

    return jsonOut_({
      found:           true,
      study_complete:  studyComplete,
      resume_snapshot: resumeSnap
    });
  } finally {
    lock.releaseLock();
  }
}

function doPost(e) {
  var body       = JSON.parse(e.postData.contents);
  var pid        = body.pid;
  if (!pid) return jsonOut_({ ok: false, error: 'missing pid' });

  var now        = new Date().toISOString();
  var sl         = body.study_level || {};
  var sections   = body.sections    || [];
  var study_id   = body.study_id    || '';
  var session_id = body.session_id  || '';

  // ── progress (non-blocking) ──────────────────────────────────
  if (body.action === 'progress') {
    var lock    = LockService.getScriptLock();
    var gotLock = lock.tryLock(1000);
    if (!gotLock) return jsonOut_({ ok: true, skipped: true });
    try {
      var metaSheet = getMetaSheet_();
      var respSheet = getRespSheet_();
      upsertMeta_(metaSheet, pid, study_id, session_id, sl, body.snapshot, now, false);
      upsertRespRows_(respSheet, pid, study_id, session_id, sl.demo_mode, sections);
      return jsonOut_({ ok: true });
    } catch (err) {
      return jsonOut_({ ok: false, error: String(err) });
    } finally {
      lock.releaseLock();
    }
  }

  // ── complete (blocking, idempotent) ─────────────────────────
  if (body.action === 'complete') {
    var lock2 = LockService.getScriptLock();
    lock2.waitLock(10000);
    try {
      var metaSheet2 = getMetaSheet_();
      var respSheet2 = getRespSheet_();

      // Idempotent: already complete → return ok without re-writing
      var existingRow = findMetaRow_(metaSheet2, pid);
      if (existingRow) {
        var existingMeta = readMetaRow_(metaSheet2, existingRow);
        if (existingMeta[META_HEADER.indexOf('study_complete_at')]) {
          return jsonOut_({ ok: true, already_complete: true });
        }
      }

      upsertMeta_(metaSheet2, pid, study_id, session_id, sl, null, now, true);
      upsertRespRows_(respSheet2, pid, study_id, session_id, sl.demo_mode, sections);
      return jsonOut_({ ok: true });
    } catch (err) {
      return jsonOut_({ ok: false, error: String(err) });
    } finally {
      lock2.releaseLock();
    }
  }

  return jsonOut_({ ok: false, error: 'unknown action' });
}

function jsonOut_(obj) {
  return ContentService.createTextOutput(JSON.stringify(obj))
    .setMimeType(ContentService.MimeType.JSON);
}
